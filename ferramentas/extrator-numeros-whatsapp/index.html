Ligar a ferramenta na sua planilha

Faz-se uma vez só, leva uns três minutos. Depois disso o botão Enviar pro Sheets manda os números direto pra planilha, sem repetir o que já está lá, e com a origem e a data de cada lote.

passo 1

Crie a planilha

  1. Abra uma planilha nova no Google Sheets.
  2. Copie o ID dela: é o pedaço da URL entre /d/ e /edit.
passo 2

Cole o script

  1. Na planilha: Extensões → Apps Script.
  2. Apague o que estiver lá e cole o código abaixo.
  3. Troque o ID_DA_PLANILHA e a SENHA nas duas primeiras linhas.
passo 3

Publique

  1. Implantar → Nova implantação → tipo App da Web.
  2. Executar como Eu, acesso para Qualquer pessoa.
  3. Autorize quando o Google pedir e copie a URL que termina em /exec.
  4. Cole essa URL e a senha nos campos aqui de cima.
Código do Apps Script 
const ID_DA_PLANILHA = 'cole-o-id-da-planilha-aqui';
const SENHA = 'troque-esta-senha';
const ABA = 'Contatos';

function doPost(e) {
  try {
    const dados = JSON.parse(e.postData.contents);
    if (dados.senha !== SENHA) return responder({ ok: false, erro: 'senha invalida' });

    const planilha = SpreadsheetApp.openById(ID_DA_PLANILHA);
    let aba = planilha.getSheetByName(ABA);
    if (!aba) aba = planilha.insertSheet(ABA);
    if (aba.getLastRow() === 0) {
      aba.appendRow(['Numero', 'DDD', 'Tipo', 'Ocorrencias', 'Origem', 'Data']);
      aba.getRange('A:A').setNumberFormat('@');
    }

    const linhas = dados.linhas || [];
    const ultima = aba.getLastRow();
    const jaTem = new Set(
      ultima > 1
        ? aba.getRange(2, 1, ultima - 1, 1).getValues().map(function (l) { return String(l[0]); })
        : []
    );

    const novas = linhas.filter(function (l) { return !jaTem.has(String(l[0])); });
    if (novas.length) {
      aba.getRange(ultima + 1, 1, novas.length, novas[0].length).setValues(novas);
    }

    return responder({ ok: true, recebidos: linhas.length, novos: novas.length });
  } catch (erro) {
    return responder({ ok: false, erro: String(erro) });
  }
}

function responder(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

A senha existe porque o app precisa ficar aberto para qualquer pessoa conseguir receber o envio do navegador. Ela impede que alguém que descubra a URL escreva na planilha. Trate a URL como interna e não publique em lugar nenhum.

Se ao publicar aparecer um aviso de app não verificado, é esperado — é um script seu, na sua conta. Clique em Avançado e siga.

conta inteira

Varrer o WhatsApp Web

É o único jeito de pegar a conta inteira de uma vez, sem exportar conversa por conversa. São dois scripts abaixo: o rápido varre só a lista; o completo abre uma conversa por vez e alcança também quem está salvo pelo nome.

Script 1 — varredura rápida da lista de conversas

No computador, abra web.whatsapp.com com a conta conectada. Aperte F12 (ou Ctrl+Shift+J / Cmd+Option+J), vá na aba Console, cole o script e dê Enter. Em dois ou três minutos ele rola a lista inteira, separa quem já aparece com número de quem está salvo pelo nome, e copia o resultado.

(async () => {
  const pane = document.querySelector('#pane-side');
  if (!pane) return console.log('Abra o WhatsApp Web antes de rodar.');

  const titulos = new Set();
  const coletar = () => document
    .querySelectorAll('#pane-side [role="listitem"] span[title]')
    .forEach(el => titulos.add(el.getAttribute('title')));

  let anterior = -1;
  for (let i = 0; i < 800 && pane.scrollTop !== anterior; i++) {
    anterior = pane.scrollTop;
    coletar();
    pane.scrollTop += pane.clientHeight * 0.8;
    await new Promise(r => setTimeout(r, 400));
  }
  coletar();

  const lista = [...titulos];
  const temNumero = /\d[\d\s()+-]{7,}\d/;
  const comNumero = lista.filter(t => temNumero.test(t));
  const salvos    = lista.filter(t => !temNumero.test(t));

  console.log('Conversas encontradas:', lista.length);
  console.log('Já vêm com número:', comNumero.length);
  console.log('Salvos na agenda, só com o nome:', salvos.length);
  console.log(comNumero.join('\n'));

  window.__numeros = comNumero.join('\n');
  window.__salvos  = salvos.join('\n');
  copy(window.__numeros);
})();

Ele te dá de cara o tamanho do trabalho: quantas conversas existem e quantas estão salvas na agenda. Os salvos aparecem pelo nome e não saem por aqui — pra eles, use o script 2 ou a exportação da agenda.

Script 2 — alcança também os contatos salvos

Este abre uma conversa por vez e lê o identificador interno que o WhatsApp guarda em cada mensagem — é de lá que sai o número mesmo quando o contato está salvo pelo nome. Leva cerca de um segundo por conversa, então deixe rodando com a aba aberta e a tela ligada. Ele só lê a tela: não envia mensagem, não abre mídia, não marca nada como lido além de abrir a conversa.

Teste antes: deixe LIMITE = 20 na primeira rodada. Se os vinte primeiros vierem certos, troque para 0 e rode a lista inteira.

(async () => {
  const LIMITE = 20;   // 0 = todas as conversas
  const PAUSA  = 900;  // ms de espera por conversa; aumente se vier vazio

  const pane = document.querySelector('#pane-side');
  if (!pane) return console.log('Abra o WhatsApp Web antes de rodar.');
  const espera = ms => new Promise(r => setTimeout(r, ms));

  const vistos  = new Set();
  const achados = new Map();   // numero -> rótulo da conversa

  const coletarDoChat = (rotulo) => {
    document.querySelectorAll('#main [data-id]').forEach(el => {
      const id = el.getAttribute('data-id') || '';
      const re = /(\d{7,15})@c\.us/g;
      let m;
      while ((m = re.exec(id))) if (!achados.has(m[1])) achados.set(m[1], rotulo);
    });
  };

  let vazias = 0;
  while (vazias < 3) {
    const itens = [...document.querySelectorAll('#pane-side [role="listitem"]')];
    let novos = 0;

    for (const li of itens) {
      const t = li.querySelector('span[title]');
      const rotulo = t ? t.getAttribute('title') : null;
      if (!rotulo || vistos.has(rotulo)) continue;
      vistos.add(rotulo);
      novos++;

      const alvo = li.querySelector('[role="gridcell"]') || li;
      alvo.dispatchEvent(new MouseEvent('mousedown', { bubbles: true }));
      alvo.dispatchEvent(new MouseEvent('mouseup',   { bubbles: true }));
      alvo.click();

      await espera(PAUSA);
      coletarDoChat(rotulo);
      console.log(vistos.size, '·', rotulo, '→', achados.size, 'números');

      if (LIMITE && vistos.size >= LIMITE) { vazias = 99; break; }
    }

    if (vazias === 99) break;
    vazias = novos ? 0 : vazias + 1;
    pane.scrollTop += pane.clientHeight * 0.85;
    await espera(700);
  }

  const texto = [...achados].map(([n, nome]) => n + '\t' + nome).join('\n');
  window.__numeros = texto;
  console.log(texto);
  console.log('TOTAL:', achados.size, 'números em', vistos.size, 'conversas');
  copy(texto);
})();

A saída vem como número + nome, uma linha por contato — cole no campo de entrada aqui em cima que a ferramenta valida, tira repetido e padroniza. Se vier vazio, aumente a PAUSA para 1500. Em grupo, ele também recolhe os participantes que já mandaram mensagem.
